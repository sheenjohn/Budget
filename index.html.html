Link Google Sheets
  1. Create a Google Sheet, then open Extensions → Apps Script.
  2. Paste the script below and choose Deploy → New deployment → Web app (access: Anyone).
  3. Paste the Web app URL here and select Sync now.
function doPost(e){
  var d=JSON.parse(e.postData.contents);
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  var s=ss.getSheetByName(d.month)||ss.insertSheet(d.month);
  s.clear();
  s.appendRow(['Category','Item','Amount','Share %']);
  d.rows.forEach(function(r){s.appendRow(r)});
  s.appendRow(['','Total',d.total,100]);
  if(d.income!==null){s.appendRow(['','Net income',d.income,'']);
    s.appendRow(['','Surplus',d.income-d.total,'']);}
  return ContentService.createTextOutput('ok');
}
